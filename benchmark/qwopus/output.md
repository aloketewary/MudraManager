### The Code

```dart
import 'dart:async';
import 'dart:math';

import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:currency_text/currency_text.dart';
import 'package:isar/isar.dart';
import 'package:lucide_icons/lucide_icons.dart';
import 'package:riverpod_annotation/riverpod_annotation.dart';
import 'package:app_localizations/app_localizations.dart';

part 'spacing_budget_card.g.dart'; // Generated by riverpod_annotation_builder

// 1. Define the Data Model (Isar)
part 'spacing_budget_card.dart'; // Generated by isar_generator

// 2. Define the Riverpod Provider
@riverpod
class SpacingBudgetCard extends _$SpacingBudgetCard {
  @override
  Stream<SpacingBudgetCard> build() async* {
    // Simulate an Isar stream or fetch logic here. 
    // For this example, we'll just fetch one item and stream it.
    final initialItem = await spacingBudgetCardCollection.get(1);
    yield initialItem;

    // Listen to changes (simulated here with a timer to show interactivity)
    final streamSubscription = spacingBudgetCardCollection.watch(
      spacingBudgetCardId1,
    ).listen((value) {
      state = value;
    });

    // Cleanup subscription when widget is disposed
    yield* streamSubscription.asStream().asStream();
  }
}

// 3. The ConsumerWidget
class SpacingBudgetCardWidget extends ConsumerWidget {
  const SpacingBudgetCardWidget({super.key});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final spacingBudgetCard = ref.watch(spacingBudgetCardProvider);

    // Access AppLocalizations safely
    final appLocalizations = AppLocalizations.of(context);

    // Calculate the remaining amount for the Hero Number
    final remainingAmount = spacingBudgetCard.remainingAmount;

    return SizedBox(
      width: double.infinity,
      child: Card(
        elevation: 0,
        child: Padding(
          padding: const EdgeInsets.all(16.0),
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              // Header
              Row(
                mainAxisAlignment: MainAxisAlignment.spaceBetween,
                children: [
                  Text(
                    appLocalizations.budgetCardTitle,
                    style: const TextStyle(
                      fontSize: 18,
                      fontWeight: FontWeight.w600,
                    ),
                  ),
                  const Icon(
                    LucideIcons.pencil,
                    color: Color(0xFF888888),
                    size: 18,
                  ),
                ],
              ),

              // Hero Number (Remaining Amount)
              // Using CurrencyText for the amount
              SizedBox(
                height: 100,
                child: Column(
                  mainAxisAlignment: MainAxisAlignment.center,
                  crossAxisAlignment: CrossAxisAlignment.end,
                  children: [
                    Text(
                      appLocalizations.remainingBudgetLabel,
                      style: const TextStyle(
                        fontSize: 14,
                        color: Color(0xFF888888),
                      ),
                    ),
                    const SizedBox(height: 4),
                    // CurrencyText formats the number with currency symbols
                    CurrencyText(
                      amount: remainingAmount,
                      locale: appLocalizations.locale,
                      format: CurrencyTextFormat.currency,
                      textTheme: const TextTheme(
                        bodyLarge: TextStyle(
                          fontSize: 48,
                          fontWeight: FontWeight.bold,
                          color: remainingAmount >= 0
                              ? Color(0xFF22C55E) // Green for positive
                              : Color(0xFFEF4444), // Red for negative
                        ),
                      ),
                    ),
                  ],
                ),
              ),

              const SizedBox(height: 24),

              // List of Transactions
              Expanded(
                child: ListView.separated(
                  padding: const EdgeInsets.symmetric(vertical: 4),
                  itemCount: spacingBudgetCard.transactions.length,
                  separatorBuilder: (context, index) => const SizedBox(height: 8),
                  itemBuilder: (context, index) {
                    final transaction = spacingBudgetCard.transactions[index];
                    final isPositive = transaction.amount >= 0;

                    return Padding(
                      padding: const EdgeInsets.symmetric(horizontal: 8),
                      child: Row(
                        mainAxisAlignment: MainAxisAlignment.spaceBetween,
                        children: [
                          // Description
                          Text(
                            transaction.description,
                            style: const TextStyle(fontSize: 14),
                          ),
                          // Amount
                          CurrencyText(
                            amount: transaction.amount,
                            locale: appLocalizations.locale,
                            format: CurrencyTextFormat.currency,
                            textTheme: const TextTheme(
                              bodyLarge: TextStyle(
                                fontSize: 14,
                                color: isPositive
                                    ? Color(0xFF888888)
                                    : Color(0xFFEF4444),
                              ),
                            ),
                          ),
                        ],
                      ),
                    );
                  },
                ),
              ),
            ],
          ),
        ),
      ),
    );
  }
}
```

### Explanation of Key Components

1.  **`ref.watch(spacingBudgetCardProvider)`**: This listens to the stream returned by the Riverpod provider. Any change in 
the Isar data triggers a rebuild of this widget.
2.  **`CurrencyText`**: Used for the hero number and transaction amounts. It automatically handles locale-specific formatting 
(e.g., adding the $ or € symbol and the correct decimal separator).
3.  **`AppLocalizations`**: Used for the strings like "Remaining Budget" and the card title. This ensures the UI respects the 
user's system locale.
4.  **`LucideIcons`**: Used for the edit icon in the header.
5.  **Hero Number Logic**: The `remainingAmount` is calculated dynamically based on the Isar data. If the remaining amount is 
positive, it displays in green; if negative, it displays in red, providing immediate visual feedback.
